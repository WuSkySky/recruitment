cmake_minimum_required(VERSION 3.8)
project(recruitment_sim_robot_base)
find_package(ament_cmake REQUIRED)
find_package(recruitment_sim_interfaces REQUIRED)
add_library(${PROJECT_NAME} INTERFACE)
target_include_directories(${PROJECT_NAME} INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS ${PROJECT_NAME} EXPORT ${PROJECT_NAME})
install(PROGRAMS scripts/test_light_color.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_enable_state test/test_enable_state.cpp)
  target_include_directories(test_enable_state PRIVATE include)
  ament_target_dependencies(test_enable_state recruitment_sim_interfaces)
  ament_add_gtest(test_gaussian_noise test/test_gaussian_noise.cpp)
  target_include_directories(test_gaussian_noise PRIVATE include)
  ament_target_dependencies(test_gaussian_noise recruitment_sim_interfaces)
  ament_add_gtest(test_incremental_odometry test/test_incremental_odometry.cpp)
  target_include_directories(test_incremental_odometry PRIVATE include)
  ament_target_dependencies(test_incremental_odometry recruitment_sim_interfaces)
  ament_add_gtest(test_initialization_gate test/test_initialization_gate.cpp)
  target_include_directories(test_initialization_gate PRIVATE include)
  ament_target_dependencies(test_initialization_gate recruitment_sim_interfaces)
endif()
ament_export_targets(${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_dependencies(recruitment_sim_interfaces)
ament_package()
